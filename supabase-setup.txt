--   • Expeditions and their codes, managed by Matan from tours.mw.photography/admin (no file uploads)
--   • Travelers join with the expedition code + their name. On any other browser or phone, the same
--     code + name brings back everything: the expedition opens, lists and field progress are restored.
--   • Photo uploads for the "animals to look for" and "photo exercises", visible to everyone in the
--     same expedition (and only to them). Each traveler can delete their own photos; Matan can delete any.
--   • Activity for Matan: who joined which expedition, when, how often, how many photos.
--
-- HOW TO RUN
--   1. Do the dashboard steps at the BOTTOM of this file first (anonymous sign-ins + Matan's admin login).
--   2. Supabase → SQL Editor → New query → paste this whole file → Run. Safe to run more than once.
--   3. Run the one-line "make Matan an admin" query (also at the bottom).

create extension if not exists pgcrypto with schema extensions;

-- ---------------------------------------------------------------------
-- 1. TABLES
-- ---------------------------------------------------------------------

-- Who may manage expeditions (Matan). Filled by the query at the bottom.
create table if not exists public.tour_admins (
  user_id  uuid primary key references auth.users (id) on delete cascade,
  email    text
);

-- One row per expedition. join_code_hash = the code travelers type (stored scrambled, never readable).
-- content_secret = the key the expedition page is locked with; handed only to travelers who joined.
create table if not exists public.tours (
  slug            text primary key check (slug ~ '^[a-z0-9][a-z0-9-]{1,63}$'),
  title           text,
  join_code_hash  text,
  content_secret  text,
  updated_at      timestamptz not null default now()
);

-- A traveler in an expedition, known by their name (case and extra spaces ignored).
create table if not exists public.members (
  id          uuid primary key default gen_random_uuid(),
  tour_slug   text not null references public.tours (slug) on delete cascade,
  name        text not null check (char_length(btrim(name)) between 1 and 60),
  name_key    text not null,
  created_at  timestamptz not null default now(),
  last_seen   timestamptz not null default now(),
  visits      integer not null default 1,
  unique (tour_slug, name_key)
);

-- Each browser/phone gets its own silent session (anonymous sign-in); joining links it to the member.
create table if not exists public.member_devices (
  user_id    uuid not null references auth.users (id) on delete cascade,
  member_id  uuid not null references public.members (id) on delete cascade,
  linked_at  timestamptz not null default now(),
  primary key (user_id, member_id)
);

-- A traveler's lists and field progress for that expedition.
create table if not exists public.progress (
  member_id   uuid primary key references public.members (id) on delete cascade,
  state       jsonb not null default '{}'::jsonb check (pg_column_size(state) < 131072),
  updated_at  timestamptz not null default now()
);

-- Photos travelers share for an animal or a photo exercise.
create table if not exists public.photos (
  id          bigint generated always as identity primary key,
  tour_slug   text not null references public.tours (slug) on delete cascade,
  member_id   uuid not null references public.members (id) on delete cascade,
  item_kind   text not null check (item_kind in ('animal', 'exercise')),
  item_id     text not null check (char_length(item_id) between 1 and 80),
  path        text not null unique,
  created_at  timestamptz not null default now()
);
create index if not exists photos_tour_idx on public.photos (tour_slug, created_at desc);

-- Wrong-code attempts, to slow down guessing.
create table if not exists public.join_attempts (
  id       bigint generated always as identity primary key,
  user_id  uuid,
  at       timestamptz not null default now()
);
create index if not exists join_attempts_user_idx on public.join_attempts (user_id, at desc);

create or replace function public.norm_code(c text) returns text
language sql immutable as $$ select upper(regexp_replace(coalesce(c, ''), '[\s\-–_.]', '', 'g')) $$;

create or replace function public.is_tour_admin() returns boolean
language sql stable security definer set search_path = public as $$
  select exists (select 1 from public.tour_admins where user_id = auth.uid())
$$;

-- the member this browser is signed in as, in a given expedition
create or replace function public.my_member(p_slug text) returns uuid
language sql stable security definer set search_path = public as $$
  select m.id from public.members m join public.member_devices d on d.member_id = m.id
  where d.user_id = auth.uid() and m.tour_slug = p_slug limit 1
$$;

-- every expedition this browser has joined
create or replace function public.my_tours() returns setof text
language sql stable security definer set search_path = public as $$
  select m.tour_slug from public.members m join public.member_devices d on d.member_id = m.id where d.user_id = auth.uid()
$$;

-- join with code + name: returns the key that opens the page, plus saved progress
create or replace function public.join_tour(p_slug text, p_code text, p_name text) returns json
language plpgsql security definer set search_path = public, extensions as $$
declare t public.tours; v_member uuid; v_name text; v_state jsonb;
begin
  if auth.uid() is null then raise exception 'no_session'; end if;
  if (select count(*) from public.join_attempts where user_id = auth.uid() and at > now() - interval '15 minutes') >= 10 then
    return json_build_object('error', 'too_many_attempts');
  end if;
  select * into t from public.tours where slug = p_slug;
  if not found or t.join_code_hash is null then return json_build_object('error', 'tour_not_ready'); end if;
  if extensions.crypt(public.norm_code(p_code), t.join_code_hash) <> t.join_code_hash then
    insert into public.join_attempts (user_id) values (auth.uid());   -- kept: an error return does not undo it
    return json_build_object('error', 'wrong_code');
  end if;
  v_name := regexp_replace(btrim(coalesce(p_name, '')), '\s+', ' ', 'g');
  if char_length(v_name) < 1 or char_length(v_name) > 60 then return json_build_object('error', 'bad_name'); end if;

  insert into public.members (tour_slug, name, name_key) values (p_slug, v_name, lower(v_name))
  on conflict (tour_slug, name_key) do update set last_seen = now(), visits = public.members.visits + 1
  returning id into v_member;
  insert into public.member_devices (user_id, member_id) values (auth.uid(), v_member) on conflict do nothing;
  insert into public.progress (member_id) values (v_member) on conflict do nothing;
  select state into v_state from public.progress where member_id = v_member;
  select name into v_name from public.members where id = v_member;
  return json_build_object('member_id', v_member, 'name', v_name, 'secret', t.content_secret, 'state', v_state);
end $$;

-- already joined on this browser: open straight away, with the latest progress
create or replace function public.resume_tour(p_slug text) returns json
language plpgsql security definer set search_path = public as $$
declare v_member uuid; t public.tours; v_name text; v_state jsonb;
begin
  v_member := public.my_member(p_slug);
  if v_member is null then return null; end if;
  select * into t from public.tours where slug = p_slug;
  update public.members set last_seen = now(),
         visits = visits + case when last_seen < now() - interval '30 minutes' then 1 else 0 end
   where id = v_member returning name into v_name;
  select state into v_state from public.progress where member_id = v_member;
  return json_build_object('member_id', v_member, 'name', v_name, 'secret', t.content_secret, 'state', v_state);
end $$;

create or replace function public.save_progress(p_slug text, p_state jsonb) returns void
language plpgsql security definer set search_path = public as $$
declare v_member uuid;
begin
  v_member := public.my_member(p_slug);
  if v_member is null then raise exception 'not_member'; end if;
  insert into public.progress (member_id, state, updated_at) values (v_member, coalesce(p_state, '{}'::jsonb), now())
  on conflict (member_id) do update set state = excluded.state, updated_at = now();
end $$;

-- 5. WHAT MATAN'S ADMIN PAGE CALLS
-- ---------------------------------------------------------------------

-- connect an expedition / change its code. p_secret: the code the page is locked with (only needed the first time).
create or replace function public.admin_set_tour(p_slug text, p_title text, p_secret text, p_code text) returns void
language plpgsql security definer set search_path = public, extensions as $$
begin
  if not public.is_tour_admin() then raise exception 'not_admin'; end if;
  if p_code is not null and char_length(public.norm_code(p_code)) < 4 then raise exception 'code_too_short'; end if;
  insert into public.tours (slug, title, content_secret, join_code_hash)
  values (p_slug, p_title, nullif(public.norm_code(p_secret), ''),
          case when p_code is null then null else extensions.crypt(public.norm_code(p_code), extensions.gen_salt('bf')) end)
  on conflict (slug) do update set
    title          = coalesce(excluded.title, public.tours.title),
    content_secret = coalesce(excluded.content_secret, public.tours.content_secret),
    join_code_hash = coalesce(excluded.join_code_hash, public.tours.join_code_hash),
    updated_at     = now();
end $$;

-- everything for the admin page in one go
create or replace function public.admin_overview() returns json
language plpgsql stable security definer set search_path = public, extensions as $$
begin
  if not public.is_tour_admin() then raise exception 'not_admin'; end if;
  return coalesce((
    select json_agg(json_build_object(
      'slug', t.slug, 'title', t.title, 'ready', t.join_code_hash is not null and t.content_secret is not null, 'updated_at', t.updated_at,
      'code', case when t.join_code_hash is not null and extensions.crypt(t.content_secret, t.join_code_hash) = t.join_code_hash then t.content_secret end,
      'members', coalesce((select json_agg(json_build_object(
          'id', m.id, 'name', m.name, 'joined', m.created_at, 'last_seen', m.last_seen, 'visits', m.visits,
          'photos', (select count(*) from public.photos p where p.member_id = m.id)) order by m.last_seen desc)
        from public.members m where m.tour_slug = t.slug), '[]'::json)
    ) order by t.slug) from public.tours t), '[]'::json);
end $$;

create or replace function public.admin_remove_member(p_member uuid) returns void
language plpgsql security definer set search_path = public as $$
begin
  if not public.is_tour_admin() then raise exception 'not_admin'; end if;
  delete from public.members where id = p_member;   -- their devices, progress and photo records go with them
end $$;

  to authenticated;

create or replace function public.view_tour(p_slug text, p_code text) returns json
language plpgsql security definer set search_path = public, extensions as $$
declare t public.tours;
begin
  if auth.uid() is null then raise exception 'no_session'; end if;
  if (select count(*) from public.join_attempts where user_id = auth.uid() and at > now() - interval '15 minutes') >= 10 then
    return json_build_object('error', 'too_many_attempts');
  end if;
  select * into t from public.tours where slug = p_slug;

-- the words of a name, lower case: 'Jeinna Louise Perez' → {jeinna, louise, perez}
create or replace function public.name_words(p text) returns text[]
language sql immutable set search_path = public as $$
  select coalesce(array_agg(w), '{}'::text[])
    from regexp_split_to_table(lower(btrim(coalesce(p, ''))), '[\s,.\-–_]+') w where w <> ''
$$;

-- 2. Joining: the code first, then the name has to be on the list
create or replace function public.join_tour(p_slug text, p_code text, p_name text) returns json
language plpgsql security definer set search_path = public, extensions as $$
declare t public.tours; v_member uuid; v_name text; v_state jsonb; v_trav uuid; v_full text; v_n int;
begin
  if auth.uid() is null then raise exception 'no_session'; end if;
  if (select count(*) from public.join_attempts where user_id = auth.uid() and at > now() - interval '15 minutes') >= 10 then
    return json_build_object('error', 'too_many_attempts');
  end if;
  select * into t from public.tours where slug = p_slug;
  if not found or t.join_code_hash is null then return json_build_object('error', 'tour_not_ready'); end if;
  if extensions.crypt(public.norm_code(p_code), t.join_code_hash) <> t.join_code_hash then
    insert into public.join_attempts (user_id) values (auth.uid());
    return json_build_object('error', 'wrong_code');
  end if;
  v_name := regexp_replace(btrim(coalesce(p_name, '')), '\s+', ' ', 'g');
  if char_length(v_name) < 1 or char_length(v_name) > 60 or cardinality(public.name_words(v_name)) = 0 then
    return json_build_object('error', 'bad_name');
  end if;

  if exists (select 1 from public.tour_travelers where tour_slug = p_slug) then
    -- every word typed must be one of the words of a single name on the list
    select count(*), min(id::text)::uuid into v_n, v_trav from public.tour_travelers
     where tour_slug = p_slug and public.name_words(v_name) <@ public.name_words(name);
    if v_n = 0 then
      insert into public.join_attempts (user_id) values (auth.uid());
      return json_build_object('error', 'name_not_listed');
    end if;
    if v_n > 1 then return json_build_object('error', 'name_ambiguous'); end if;
    select name into v_full from public.tour_travelers where id = v_trav;
    select id into v_member from public.members where tour_slug = p_slug and traveler_id = v_trav limit 1;
    if v_member is null then
      insert into public.members (tour_slug, name, name_key, traveler_id) values (p_slug, v_full, lower(v_full), v_trav)
      on conflict (tour_slug, name_key) do update set traveler_id = excluded.traveler_id, last_seen = now(), visits = public.members.visits + 1
      returning id into v_member;
    else
      update public.members set last_seen = now(), visits = visits + 1 where id = v_member;
    end if;
  else
    insert into public.members (tour_slug, name, name_key) values (p_slug, v_name, lower(v_name))
    on conflict (tour_slug, name_key) do update set last_seen = now(), visits = public.members.visits + 1
    returning id into v_member;
  end if;

  insert into public.member_devices (user_id, member_id) values (auth.uid(), v_member) on conflict do nothing;
  insert into public.progress (member_id) values (v_member) on conflict do nothing;
  select state into v_state from public.progress where member_id = v_member;
  select name into v_name from public.members where id = v_member;
  return json_build_object('member_id', v_member, 'name', v_name, 'secret', t.content_secret, 'state', v_state);
end $$;

-- 3. Admin: add names (several at once), remove a name
create or replace function public.admin_add_travelers(p_slug text, p_names text[]) returns int
language plpgsql security definer set search_path = public as $$
declare n text; v_added int := 0; v_rows int;
begin
  if not public.is_tour_admin() then raise exception 'not_admin'; end if;
  if not exists (select 1 from public.tours where slug = p_slug) then raise exception 'tour_not_connected'; end if;
  foreach n in array coalesce(p_names, '{}'::text[]) loop
    n := regexp_replace(btrim(coalesce(n, '')), '\s+', ' ', 'g');
    continue when char_length(n) < 1 or char_length(n) > 60;
    insert into public.tour_travelers (tour_slug, name, name_key) values (p_slug, n, lower(n)) on conflict do nothing;
    get diagnostics v_rows = row_count; v_added := v_added + v_rows;
  end loop;
  return v_added;
end $$;

-- removing a name also removes that traveler's access, progress and photo records
create or replace function public.admin_remove_traveler(p_id uuid) returns void
language plpgsql security definer set search_path = public as $$
begin
  if not public.is_tour_admin() then raise exception 'not_admin'; end if;
  delete from public.members where traveler_id = p_id;
  delete from public.tour_travelers where id = p_id;
end $$;

-- 4. The admin page also gets the list
create or replace function public.admin_overview() returns json
language plpgsql stable security definer set search_path = public, extensions as $$
begin
  if not public.is_tour_admin() then raise exception 'not_admin'; end if;
  return coalesce((
    select json_agg(json_build_object(
      'slug', t.slug, 'title', t.title, 'ready', t.join_code_hash is not null and t.content_secret is not null, 'updated_at', t.updated_at,
      'code', case when t.join_code_hash is not null and extensions.crypt(t.content_secret, t.join_code_hash) = t.join_code_hash then t.content_secret end,
      'travelers', coalesce((select json_agg(json_build_object('id', r.id, 'name', r.name) order by lower(r.name))
        from public.tour_travelers r where r.tour_slug = t.slug), '[]'::json),
      'members', coalesce((select json_agg(json_build_object(
          'id', m.id, 'name', m.name, 'traveler_id', m.traveler_id, 'joined', m.created_at, 'last_seen', m.last_seen, 'visits', m.visits,
          'photos', (select count(*) from public.photos p where p.member_id = m.id)) order by m.last_seen desc)
        from public.members m where m.tour_slug = t.slug), '[]'::json)
    ) order by t.slug) from public.tours t), '[]'::json);
end $$;

revoke execute on function public.name_words(text), public.admin_add_travelers(text, text[]), public.admin_remove_traveler(uuid) from public, anon;
grant execute on function public.admin_add_travelers(text, text[]), public.admin_remove_traveler(uuid) to authenticated;

-- ---------------------------------------------------------------------
-- 9. MATAN OPENS A TRAVELER'S VIEW (read only)
-- ---------------------------------------------------------------------

create or replace function public.admin_view_member(p_member uuid) returns json
language plpgsql stable security definer set search_path = public as $$
declare m public.members; r public.tour_travelers; t public.tours; v_state jsonb;
begin
  if not public.is_tour_admin() then raise exception 'not_admin'; end if;
  select * into m from public.members where id = p_member;
  if not found then
    -- a name on the list: their own record if they joined, otherwise a fresh view
    select * into r from public.tour_travelers where id = p_member;
    if not found then return null; end if;
    select * into m from public.members where traveler_id = r.id limit 1;
    if not found then
      select * into t from public.tours where slug = r.tour_slug;
      return json_build_object('member_id', null, 'traveler_id', r.id, 'name', r.name, 'tour_slug', r.tour_slug,
                               'secret', t.content_secret, 'state', '{}'::jsonb);
    end if;
  end if;
  select * into t from public.tours where slug = m.tour_slug;
  select state into v_state from public.progress where member_id = m.id;
  return json_build_object('member_id', m.id, 'traveler_id', m.traveler_id, 'name', m.name, 'tour_slug', m.tour_slug,
                           'secret', t.content_secret, 'state', coalesce(v_state, '{}'::jsonb));
end $$;

revoke execute on function public.admin_view_member(uuid) from public, anon;
grant execute on function public.admin_view_member(uuid) to authenticated;

create or replace function public.join_tour(p_slug text, p_code text, p_name text) returns json
language plpgsql security definer set search_path = public, extensions as $$
declare t public.tours; v_member uuid; v_name text; v_state jsonb; v_trav uuid; v_full text; v_n int;
begin
  if auth.uid() is null then raise exception 'no_session'; end if;
  if (select count(*) from public.join_attempts where user_id = auth.uid() and at > now() - interval '15 minutes') >= 10 then
    return json_build_object('error', 'too_many_attempts');
  end if;
  select * into t from public.tours where slug = p_slug;
  if not found or t.join_code_hash is null then return json_build_object('error', 'tour_not_ready'); end if;
  if extensions.crypt(public.norm_code(p_code), t.join_code_hash) <> t.join_code_hash then
    insert into public.join_attempts (user_id) values (auth.uid());
    return json_build_object('error', 'wrong_code');
  end if;
  v_name := regexp_replace(btrim(coalesce(p_name, '')), '\s+', ' ', 'g');
  if char_length(v_name) < 1 or char_length(v_name) > 60 or cardinality(public.name_words(v_name)) = 0 then
    return json_build_object('error', 'bad_name');
  end if;

  -- only names Matan put on the list can join; every word typed must be one of the words of a single name
  select count(*), min(id::text)::uuid into v_n, v_trav from public.tour_travelers
   where tour_slug = p_slug and public.name_words(v_name) <@ public.name_words(name);
  if v_n = 0 then
    insert into public.join_attempts (user_id) values (auth.uid());
    return json_build_object('error', 'name_not_listed');
  end if;
  if v_n > 1 then return json_build_object('error', 'name_ambiguous'); end if;
  select name into v_full from public.tour_travelers where id = v_trav;
  select id into v_member from public.members where tour_slug = p_slug and traveler_id = v_trav limit 1;
  if v_member is null then
    insert into public.members (tour_slug, name, name_key, traveler_id) values (p_slug, v_full, lower(v_full), v_trav)
    on conflict (tour_slug, name_key) do update set traveler_id = excluded.traveler_id, last_seen = now(), visits = public.members.visits + 1
    returning id into v_member;
  else
    update public.members set last_seen = now(), visits = visits + 1 where id = v_member;
  end if;

  insert into public.member_devices (user_id, member_id) values (auth.uid(), v_member) on conflict do nothing;
  insert into public.progress (member_id) values (v_member) on conflict do nothing;
  select state into v_state from public.progress where member_id = v_member;
  select name into v_name from public.members where id = v_member;
  return json_build_object('member_id', v_member, 'name', v_name, 'secret', t.content_secret, 'state', v_state);
end $$;

-- a browser that is linked to more than one traveler record: the one on the list comes first
create or replace function public.my_member(p_slug text) returns uuid
language sql stable security definer set search_path = public as $$
  select m.id from public.members m join public.member_devices d on d.member_id = m.id
  where d.user_id = auth.uid() and m.tour_slug = p_slug
  order by (m.traveler_id is null), m.last_seen desc limit 1
$$;

-- already joined on this browser: open straight away, only while their name is on the list
create or replace function public.resume_tour(p_slug text) returns json
language plpgsql security definer set search_path = public as $$
declare v_member uuid; t public.tours; v_name text; v_state jsonb;
begin
  v_member := public.my_member(p_slug);
  if v_member is null then return null; end if;
  if not exists (select 1 from public.members m join public.tour_travelers r on r.id = m.traveler_id where m.id = v_member) then
    return null;
  end if;
  select * into t from public.tours where slug = p_slug;
  update public.members set last_seen = now(),
         visits = visits + case when last_seen < now() - interval '30 minutes' then 1 else 0 end
   where id = v_member returning name into v_name;
  select state into v_state from public.progress where member_id = v_member;
  return json_build_object('member_id', v_member, 'name', v_name, 'secret', t.content_secret, 'state', v_state);
end $$;
