-- MW Photo Tours · update: deleting photos (admin any photo, travelers their own)
-- Fixes "permission denied for table member_devices" when deleting or replacing a photo.
-- Run once: Supabase → SQL Editor → New query → paste all of this → Run.
-- =====================================================================

-- the traveler records linked to this browser (read with the owner's rights, so the table stays private)
create or replace function public.my_members() returns setof uuid
language sql stable security definer set search_path = public as $$
  select d.member_id from public.member_devices d where d.user_id = auth.uid()
$$;
revoke execute on function public.my_members() from public, anon;
grant execute on function public.my_members() to authenticated;

drop policy if exists "Delete own photos, admin any" on public.photos;
create policy "Delete own photos, admin any" on public.photos for delete to authenticated
  using (member_id in (select public.my_members()) or public.is_tour_admin());

drop policy if exists "Tour members delete their own photos, admin any" on storage.objects;
create policy "Tour members delete their own photos, admin any" on storage.objects for delete to authenticated
  using (bucket_id = 'tour-photos' and ((storage.foldername(name))[2] in (select public.my_members()::text)
         or public.is_tour_admin()));
