-- already joined on another device: the name alone opens it again (first name is enough when it's unique)
create or replace function public.join_by_name(p_slug text, p_name text) returns json
language plpgsql security definer set search_path = public as $$
declare t public.tours; v_key text; v_member uuid; v_count int; v_name text; v_state jsonb;
begin
  if auth.uid() is null then raise exception 'no_session'; end if;
  if (select count(*) from public.join_attempts where user_id = auth.uid() and at > now() - interval '15 minutes') >= 10 then
    return json_build_object('error', 'too_many_attempts');
  end if;
  select * into t from public.tours where slug = p_slug;
  if not found or t.join_code_hash is null then return json_build_object('error', 'tour_not_ready'); end if;
  v_key := lower(regexp_replace(btrim(coalesce(p_name, '')), '\s+', ' ', 'g'));
  if v_key = '' then return json_build_object('error', 'bad_name'); end if;
  -- the exact name first; otherwise a unique traveler whose first name matches
  select id into v_member from public.members where tour_slug = p_slug and name_key = v_key;
  if v_member is null then
    select count(*), min(id::text)::uuid into v_count, v_member from public.members
     where tour_slug = p_slug and split_part(name_key, ' ', 1) = v_key;
    if v_count > 1 then return json_build_object('error', 'ambiguous'); end if;
  end if;
  if v_member is null then
    insert into public.join_attempts (user_id) values (auth.uid());
    return json_build_object('error', 'unknown_name');
  end if;
  update public.members set last_seen = now(), visits = visits + 1 where id = v_member returning name into v_name;
  insert into public.member_devices (user_id, member_id) values (auth.uid(), v_member) on conflict do nothing;
  insert into public.progress (member_id) values (v_member) on conflict do nothing;
  select state into v_state from public.progress where member_id = v_member;
  return json_build_object('member_id', v_member, 'name', v_name, 'secret', t.content_secret, 'state', v_state);
end $$;

grant execute on function public.join_by_name(text, text) to authenticated;
